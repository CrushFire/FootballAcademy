В.1 — Сервис аутентификации (AuthService)

public async Task<Result<AuthResponse>> AuthorizationAsync(AuthRequest req)
{
    var input = (req.Identifier ?? req.Email ?? string.Empty).Trim();
    var mode = req.Mode?.ToLowerInvariant();
    if (string.IsNullOrEmpty(input))
        return Result<AuthResponse>.Failure("Укажите логин или email", 400);

    var normalized = mode == "email" ? input.ToLowerInvariant() : input;

    User? user = mode switch
    {
        "email" => await _context.Users.FirstOrDefaultAsync(u => u.Email.ToLower() == normalized),
        "login" => await _context.Users.FirstOrDefaultAsync(u => u.Login == normalized),
    };

    if (user == null)
        return Result<AuthResponse>.Failure("Пользователь не найден", 404);

    if (!_passwordHasher.VerifyPassword(req.Password, user.Password))
        return Result<AuthResponse>.Failure("Неверный пароль", 401);

    string? personalType = null;
    string? personalId = null;
    if (user.Role == "personal")
    {
        var personal = await _context.Personal.FirstOrDefaultAsync(p => p.UserId == user.Id);
        personalType = personal?.Type.ToString();
        personalId = personal?.Id.ToString();
    }

    var token = _jwtTokenGen.TokenGen(user.Id.ToString(), user.Login,
        user.Email, user.Role, personalType, personalId);
    return Result<AuthResponse>.Success(new AuthResponse { Token = token, UserId = user.Id });
}


В.2 — Сервис пользователей (UserService)

public async Task<Result<bool>> CreateUserAsync(UserCreateRequest req)
{
    if (req == null)
        return Result<bool>.Failure("Данные пользователя не могут быть пустыми", 400);

    var existingUser = await _context.Users.FirstOrDefaultAsync(x => x.Email == req.Email);
    if (existingUser != null)
        return Result<bool>.Failure("Пользователь с таким email уже существует", 409);

    var newUser = _mapper.Map<User>(req);
    if (!string.IsNullOrWhiteSpace(req.Password))
        newUser.Password = _passwordHasher.HashPassword(req.Password);

    await _context.Users.AddAsync(newUser);
    await _context.SaveChangesAsync();
    return Result<bool>.Success(true);
}


В.3 — Сервис спортсменов (SportsmanService)

public async Task<Result<SportsmanResponse>> GetByIdAsync(long id)
{
    var sportsman = await _context.Sportsmen
        .Include(s => s.Images)
        .Include(s => s.Team)
        .Include(s => s.SportsmanGroups).ThenInclude(sg => sg.Group)
        .FirstOrDefaultAsync(s => s.Id == id);

    if (sportsman == null)
        return Result<SportsmanResponse>.Failure("Спортсмен не найден", 404);

    var response = _mapper.Map<SportsmanResponse>(sportsman);
    response.Age = CalculateAge(sportsman.BirthDate);
    return Result<SportsmanResponse>.Success(response);
}

private int CalculateAge(DateTime birthDate)
{
    var today = DateTime.Today;
    var age = today.Year - birthDate.Year;
    if (birthDate.Date > today.AddYears(-age)) age--;
    return age;
}

public async Task<Result<string>> ImportMetricsFromFileAsync(string filePath)
{
    using var stream = new FileStream(filePath, FileMode.Open, FileAccess.Read);
    var formFile = new FormFile(stream, 0, stream.Length, "file",
        Path.GetFileName(filePath));

    var parseResult = await _parser.ParseExcelFileAsync(formFile);
    if (!parseResult.IsSuccess)
        return Result<string>.Failure(parseResult.ErrorMessage, parseResult.StatusCode);

    var metrics = parseResult.Data.Select(req => _mapper.Map<TrainingMetrics>(req)).ToList();
    await _context.TrainingMetrics.AddRangeAsync(metrics);
    await _context.SaveChangesAsync();

    return Result<string>.Success($"Импортировано {parseResult.Data.Count} метрик");
}


В.9 — Сервис автоматического импорта (MetricAutoImportService)

public async Task<Result<AutoImportResult>> RunAutoImportNowAsync()
{
    const string folder = "/app/uploads/metrics";
    if (!Directory.Exists(folder)) Directory.CreateDirectory(folder);

    using var scope = _serviceProvider.CreateScope();
    var metricService = scope.ServiceProvider.GetRequiredService<IMetricsService>();
    var files = Directory.GetFiles(folder, "*.xlsx");

    int processed = 0, failed = 0, metricsImported = 0;
    string? lastError = null;

    foreach (var filePath in files)
    {
        var importResult = await metricService.ImportMetricsFromFileAsync(filePath);
        if (!importResult.IsSuccess)
        {
            failed++;
            lastError = importResult.ErrorMessage;
            continue;
        }

        var match = Regex.Match(importResult.Data ?? "", @"\d+");
        if (match.Success && int.TryParse(match.Value, out var n))
            metricsImported += n;

        var archivePath = Path.Combine(folder, "archive");
        Directory.CreateDirectory(archivePath);
        File.Move(filePath, Path.Combine(archivePath, Path.GetFileName(filePath)));
        processed++;
    }

    return Result<AutoImportResult>.Success(new AutoImportResult
    {
        Processed = processed,
        Failed = failed,
        MetricsImported = metricsImported
    });
}


В.10 — Сервис основных метрик (MainMetricService)

public CommonMetrics CalculateCommonMetrics(TrainingMetrics m)
{
    var durationMin = m.Duration / 60.0;
    return new CommonMetrics
    {
        SprintRatio = m.TotalDistance > 0
            ? (double)m.SprintDistance / m.TotalDistance : 0,
        HighSpeedRatio = m.TotalDistance > 0
            ? (double)m.HighSpeedDistance / m.TotalDistance : 0,
        AvgSpeed = durationMin > 0 ? m.TotalDistance / durationMin : 0,
        HRRedPercent = m.Duration > 0
            ? (double)m.TimeInHRRedZone / m.Duration : 0,
        AerobicLoad = m.TotalDistance > 0
            ? (double)(m.LowSpeedDistance + m.ModerateSpeedDistance) / m.TotalDistance : 0,
        CardioEfficiency = m.AverageHeartRate > 0
            ? (double)m.TotalDistance / m.AverageHeartRate : 0,
        HRStability = m.MaxHeartRate > 0
            ? (double)m.AverageHeartRate / m.MaxHeartRate : 0,
        ExplosiveIndex = m.Duration > 0
            ? (double)(m.ExplosiveEfforts + m.AccelerationCount
                + m.DecelerationCount) / m.Duration : 0
    };
}


В.11 — Сервис медицинских метрик (MedicalMetricService)

public MedicalCheckResult PerformCheck(TrainingMetrics m, int athleteAge, double acwr)
{
    var maxHrThreshold = GetMaxHeartRateBase(athleteAge) - athleteAge;
    var cvOk = m.MaxHeartRate < maxHrThreshold && m.AverageHeartRate < 180;
    var loadOk = m.PlayerLoad < 500 && acwr < 1.5;
    var recoveryOk = m.TimeInSpeedZone1 > 300;
    var injuryRiskOk = m.PlayerLoad < 450
        && (m.AccelerationCount + m.DecelerationCount) < 200 && acwr < 1.3;

    var fatigueOk = m.TimeInHRRedZone < 600;
    var issues = new List<string>();
    if (!cvOk) issues.Add("Сердечно-сосудистая система");
    if (!loadOk) issues.Add("Превышение нагрузки");
    if (!recoveryOk) issues.Add("Недостаточное восстановление");
    if (!injuryRiskOk) issues.Add("Высокий риск травмы");
    if (!fatigueOk) issues.Add("Утомление");

    return new MedicalCheckResult
    {
        IsHealthy = issues.Count == 0,
        Issues = issues
    };
}

private int GetMaxHeartRateBase(int age) => age switch
{
    <= 10 => 210,
    <= 12 => 210,
    <= 14 => 208,
    <= 16 => 206,
    <= 18 => 204,
    <= 21 => 202,
    _ => 205
};


В.12 — Сервис пятиугольника (PentagonService)

public PentagonResult CalculatePentagon(TrainingMetrics m, Sportsman s,
    List<TrainingMetrics> groupMetrics)
{
    var raw = CalculateRawPentagon(m);
    var posCoefs = _positionCoefficientService.GetByPosition(s.Position);
    var absStandards = _absoluteStandardService.GetByAgeAndGender(s.Age, s.Gender);
    var n = groupMetrics.Count;
    var (wRel, wAbs) = GetNormalizationWeights(n);

    var result = new PentagonResult();
    foreach (var axis in new[] { "Speed", "Power", "Sprints", "Endurance", "Explosive" })
    {
        var groupMax = groupMetrics.Max(gm => GetRawValue(axis, gm));
        var adjG = groupMax * posCoefs[axis];
        var adjA = absStandards[axis] * posCoefs[axis];

        var rel = adjG > 0 ? raw[axis] / adjG : 0;
        var abs = adjA > 0 ? raw[axis] / adjA : 0;
        var score = Math.Clamp(wRel * rel + wAbs * abs, 0, 1);

        result[axis] = score;
    }
    return result;
}

private (double wRel, double wAbs) GetNormalizationWeights(int n)
{
    if (n >= 30) return (0.80, 0.20);
    if (n >= 15) return (0.70, 0.30);
    if (n >= 5) return (0.50, 0.50);
    return (0.20, 0.80);
}


В.13 — Сервис профилей игрока (ProfileService)

public List<PlayerProfile> DetermineProfiles(TrainingMetrics m, Sportsman s)
{
    var profiles = new List<PlayerProfile>();
    var common = _mainMetricService.CalculateCommonMetrics(m);
    var posCoef = _positionCoefficientService.GetByPosition(s.Position);
    var durationMin = m.Duration / 60.0;
    var accPerSec = m.Duration > 0 ? (double)m.AccelerationCount / m.Duration : 0;
    var defIdx = m.TotalDistance > 0 ? m.PlayerLoad / m.TotalDistance : 0;

    if (common.SprintRatio > 0.45 && accPerSec > 0.15)
        profiles.Add(PlayerProfile.Sprinter);

    if (common.AerobicLoad > 0.35)
        profiles.Add(PlayerProfile.EnduranceRunner);

    if (m.PlayerLoad > 300)
        profiles.Add(PlayerProfile.PowerPlayer);

    if (common.ExplosiveIndex > 1.2)
        profiles.Add(PlayerProfile.ExplosivePlayer);

    if (defIdx > 2.0 && posCoef["Power"] >= 1.0)
        profiles.Add(PlayerProfile.DefenderType);

    if (profiles.Count >= 4)
        profiles.Add(PlayerProfile.Universal);

    return profiles;
}

public async Task<Result<List<GraphPoint>>> GetGraphPointsAsync(
    long sportsmanId, string metricKey, DateTime from, DateTime to)
{
    var metrics = await _context.TrainingMetrics
        .Include(tm => tm.Training)
        .Where(tm => tm.SportsmanId == sportsmanId
                  && tm.Training.Date >= from
                  && tm.Training.Date <= to)

        .ToListAsync();

        Date = tm.Training.Date,
        Value = ExtractMetricValue(tm, metricKey),
        TrainingType = tm.Training.Type,
        MatchType = tm.Training.MatchId.HasValue ? "Match" : null
    }).ToList();

    return Result<List<GraphPoint>>.Success(points);
}

В.19 — Сервис очистки матчей (MatchCleanupService)

public async Task CleanupStaleMatchesAsync()
{
    var threshold = DateTime.UtcNow.AddHours(-6);
    var staleMatches = await _context.Matches
        .Where(m => m.Status == MatchStatus.InProgress
                 && m.StartedAt != null
                 && m.StartedAt < threshold)
        .ToListAsync();

    foreach (var match in staleMatches)
    {
        match.Status = MatchStatus.Cancelled;
        match.TrainerComment = (match.TrainerComment ?? "")
            + " [Автоматически отменён: превышено время проведения]";
    }

    if (staleMatches.Count > 0)
        await _context.SaveChangesAsync();
}

В.21 — Сервис нормативов (NormativeService)

public string CalculateGrade(double result, Normative norm)
{
    var lower = norm.GradeExcellent < norm.GradeSatisfactory;
    if (lower)
    {
        if (result <= norm.GradeExcellent) return "excellent";
        if (result <= norm.GradeGood) return "good";
        if (result <= norm.GradeSatisfactory) return "satisfactory";
    }
    else
    {
        if (result >= norm.GradeExcellent) return "excellent";
        if (result >= norm.GradeGood) return "good";
        if (result >= norm.GradeSatisfactory) return "satisfactory";
    }
    return "none";
}

        <span class="tag">Система управления академией</span>
      </div>

      <div class="mode-switch">
        <button type="button" class="mode-btn"
          :class="{ active: mode === 'login' }" @click="mode = 'login'">
          По логину
        </button>
        <button type="button" class="mode-btn"
          :class="{ active: mode === 'email' }" @click="mode = 'email'">
          По email
        </button>
      </div>

      <div class="field-group">
        <label>{{ mode === 'login' ? 'Логин' : 'Email' }}</label>
        <input v-model="identifier"
          :type="mode === 'email' ? 'email' : 'text'" required />
      </div>

      <div class="field-group">
        <label>Пароль</label>
        <div class="input-wrap">
          <input v-model="password"
            :type="showPassword ? 'text' : 'password'" required />
          <span class="input-icon" @click="showPassword = !showPassword">
            {{ showPassword ? '🙈' : '👁' }}
          </span>
        </div>
      </div>

      <button type="submit" class="submit-btn" :disabled="loading">
        {{ loading ? 'Вход...' : 'Войти' }}
      </button>

      <p v-if="error" class="error">{{ error }}</p>
    </form>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import { useRouter } from 'vue-router'
import { useAuthStore } from '@/store/auth'

const router = useRouter()
const auth = useAuthStore()

const mode = ref<'login' | 'email'>('login')
const identifier = ref('')
const password = ref('')
const showPassword = ref(false)
const loading = ref(false)
const error = ref('')

async function onSubmit() {
  loading.value = true
  error.value = ''
  try {
    await auth.login({ identifier: identifier.value,
      password: password.value, mode: mode.value })
    router.push('/')
  } catch (e: any) {
    error.value = e.message ?? 'Ошибка входа'
  } finally {
    loading.value = false
  }
}
</script>
